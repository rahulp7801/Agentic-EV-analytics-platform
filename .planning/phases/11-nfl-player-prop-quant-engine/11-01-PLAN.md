```python
ALLOWED_FILTER_KEYS: frozenset[str] = frozenset({"down", "ydstogo", "defteam", "game_id", "week"})

class QueryBuilder:
    @classmethod
    def build(cls, params: QuantParams) -> tuple[str, tuple[object, ...]]:
        sql: str = _QUERY_TEMPLATES[params.stat_type]
        args: list[object] = [params.posteam, params.season]
        next_idx = 3
        for key, value in params.filters.items():
            if key not in ALLOWED_FILTER_KEYS:
                log.warning("unknown_filter_key_dropped", key=key)
                continue
            sql = sql + f"  AND {key} = ${next_idx}\n"
            args.append(value)
            next_idx += 1
        return sql, tuple(args)
```

From src/sportsbet/quant/executor.py (pattern to mirror):
```python
MIN_SAMPLE_SIZE: int = 30

async def run_quant_query(pool: asyncpg.Pool, params: QuantParams) -> QuantResult:
    sql, args = QueryBuilder.build(params)
    async with pool.acquire() as conn:
        row = await conn.fetchrow(sql, *args)
    total: int = int(row["total"]) if row and row["total"] is not None else 0
    successes: int = int(row["successes"]) if row and row["successes"] is not None else 0
    if total < MIN_SAMPLE_SIZE:
        return QuantResult(data_source="insufficient_sample", sample_size=total)
    lo, hi = proportion_confint(count=successes, nobs=total, alpha=0.05, method="wilson")
    true_prob = Decimal(str(round(successes / total, 6)))
    ci = (Decimal(str(round(lo, 6))), Decimal(str(round(hi, 6))))
    return QuantResult(true_probability=true_prob, sample_size=total, confidence_interval=ci, data_source="postgresql")
```

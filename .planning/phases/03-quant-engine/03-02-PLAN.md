American odds to raw implied probability (two branches):
```python
# Negative odds (favorites): abs(odds) / (abs(odds) + 100)
# -110 → 110 / (110 + 100) = 110/210 ≈ 0.523809...
# Positive odds (underdogs): 100 / (odds + 100)
# +150 → 100 / (150 + 100) = 100/250 = 0.400
# CRITICAL: Always construct Decimal from string, never from float.
# Decimal(str(-110)) is correct. Decimal(-110) is wrong.
```

Multiplicative devig:
```python
# overround = sum(raw_probs)
# fair_probs = [p / overround for p in raw_probs]
# Guard: if overround <= Decimal("1"): raise ValueError(...)
```

Power devig (binary search on exponent k):
```python
# find k in (0, 1] such that sum(p_i^k) == 1
# 60 iterations of binary search gives precision below 1e-9
# lo, hi = 0.0, 1.0; mid = (lo + hi) / 2; compare sum(float(p)**mid for p in raw_probs)
# Final k = Decimal(str((lo + hi) / 2)); return [p**k for p in raw_probs]
```

    ```python
    from decimal import Decimal
    import math  # for float() in power binary search only
    from typing import Sequence

    def american_to_raw_prob(american_odds: int) -> Decimal:
        """Convert American odds integer to vig-inclusive implied probability as Decimal."""
        # CRITICAL: Decimal(str(x)) not Decimal(x) — avoids float precision artifacts
        odds = Decimal(str(american_odds))
        if american_odds < 0:
            return abs(odds) / (abs(odds) + Decimal("100"))
        else:
            return Decimal("100") / (odds + Decimal("100"))

    def remove_vig_multiplicative(raw_probs: Sequence[Decimal]) -> list[Decimal]:
        """Proportional devig: normalize each prob by the overround total.

        Raises ValueError if overround <= 1 (corrupt/invalid market input).
        """
        overround = sum(raw_probs)
        if overround <= Decimal("1"):
            raise ValueError(f"Invalid market: overround {overround} <= 1. Both-positive-odds markets cannot be devigged.")
        return [p / overround for p in raw_probs]

    def remove_vig_power(raw_probs: Sequence[Decimal], tol: Decimal = Decimal("1e-9")) -> list[Decimal]:
        """Power (Pinnacle sharp) devig: find k in (0,1] such that sum(p_i^k) == 1.

        Corrects favorite-longshot bias that multiplicative method ignores.
        60 iterations of binary search gives precision well below 1e-9 tolerance.
        """
        lo, hi = 0.0, 1.0
        for _ in range(60):
            mid = (lo + hi) / 2
            total = sum(float(p) ** mid for p in raw_probs)
            if total > 1.0:
                hi = mid
            else:
                lo = mid
        k = Decimal(str((lo + hi) / 2))
        return [p ** k for p in raw_probs]
    ```

    Logic:
    1. today = target_date or date_cls.today()

    4. try: player_id = int(player_id_raw) except (ValueError, TypeError) -> log.warning + return {"nba_context_signals": None}
    5. async with pool.acquire() as conn:
       a. fetchrow: SELECT team_abbreviation, game_date FROM nba_player_gamelogs WHERE player_id=$1 AND season=$2 ORDER BY game_date DESC LIMIT 1

       c. team_abbr = row["team_abbreviation"]; last_game_date = row["game_date"]

       e. is_home = (team_abbr == home_team); opponent_abbr = away_team if is_home else home_team

```python
from __future__ import annotations
import asyncpg
import sqlalchemy as sa
from sportsbet.config import settings

def get_sync_engine() -> sa.Engine:
    """Return a SQLAlchemy sync engine (psycopg v3 driver for bulk writes)."""
    return sa.create_engine(settings.database_url, echo=False, pool_pre_ping=True)

async def create_async_pool(min_size: int = 2, max_size: int = 10) -> asyncpg.Pool:
    """Return an asyncpg connection pool for runtime quant agent queries."""
    pool: asyncpg.Pool = await asyncpg.create_pool(
        settings.database_url_async,
        min_size=min_size,
        max_size=max_size,
        command_timeout=60,
    )
    return pool
```

Create `alembic/versions/0001_initial_schema.py` as a hand-written migration (NOT autogenerated — autogenerate omits composite indexes per Pitfall 4 in RESEARCH.md).

The migration must:
1. Create tables in dependency order: games first, then tables with FKs to games
2. Call `op.create_index()` for ALL composite indexes WITHIN the same migration function — before any data could ever be loaded
3. Define `downgrade()` that drops tables in reverse FK order

Migration structure:
```python
"""initial schema

Revision ID: 0001
Revises:
Create Date: 2026-03-10

"""
from alembic import op
import sqlalchemy as sa
from sqlalchemy.dialects import postgresql

revision = "0001"
down_revision = None
branch_labels = None
depends_on = None

def upgrade() -> None:
    # 1. games (no FK dependencies)
    op.create_table("games", ...)
    op.create_index("idx_games_season_week", "games", ["season", "week"])
    op.create_index("idx_games_teams", "games", ["home_team", "away_team"])

    # 2. play_by_play
    op.create_table("play_by_play", ...)
    op.create_index("idx_pbp_season_week", "play_by_play", ["season", "week"])
    op.create_index("idx_pbp_posteam_season", "play_by_play", ["posteam", "season"])
    op.create_index("idx_pbp_defteam_season", "play_by_play", ["defteam", "season"])
    op.create_index("idx_pbp_play_type_season", "play_by_play", ["play_type", "season", "week"])
    op.create_index("idx_pbp_passer_season", "play_by_play", ["passer_player_id", "season"])
    op.create_index("idx_pbp_receiver_season", "play_by_play", ["receiver_player_id", "season"])

    # 3. player_stats
    op.create_table("player_stats", ...)
    op.create_index("idx_playerstats_player_season", "player_stats", ["player_id", "season"])
    op.create_index("idx_playerstats_season_week", "player_stats", ["season", "week"])

    # 4. ngs_stats
    op.create_table("ngs_stats", ...)
    op.create_index("idx_ngs_player_season", "ngs_stats", ["player_gsis_id", "season"])
    op.create_index("idx_ngs_season_week_type", "ngs_stats", ["season", "week", "stat_type"])

    # 5. odds_snapshots
    op.create_table("odds_snapshots", ...)
    op.create_index("idx_odds_game_market", "odds_snapshots", ["game_id", "market_type"])
    op.create_index(
        "idx_odds_snapped_at_desc", "odds_snapshots", ["snapped_at"],
        postgresql_ops={"snapped_at": "DESC"},
    )

def downgrade() -> None:
    # Drop in reverse FK order
    op.drop_table("odds_snapshots")
    op.drop_table("ngs_stats")
    op.drop_table("player_stats")
    op.drop_table("play_by_play")
    op.drop_table("games")
```

```python
def test_pbp_indexes(pg_engine) -> None:
    # Query pg_indexes system catalog for play_by_play composite indexes
    with pg_engine.connect() as conn:
        result = conn.execute(sa.text(
            "SELECT indexname FROM pg_indexes WHERE tablename = 'play_by_play'"
        ))
        index_names = {row[0] for row in result}
    required = {
        "idx_pbp_season_week", "idx_pbp_posteam_season", "idx_pbp_defteam_season",
        "idx_pbp_play_type_season", "idx_pbp_passer_season", "idx_pbp_receiver_season",
    }
    assert required.issubset(index_names), f"Missing indexes: {required - index_names}"

def test_explain_no_seq_scan(pg_engine) -> None:
    # Insert a minimal dummy row so PostgreSQL query planner has stats
    with pg_engine.begin() as conn:
        conn.execute(sa.text(
            "INSERT INTO play_by_play (game_id, play_id, season, week, posteam, play_type) "
            "VALUES ('2023_01_KC_DET', 1, 2023, 1, 'KC', 'pass') ON CONFLICT DO NOTHING"
        ))
        plan = conn.execute(sa.text(
            "EXPLAIN ANALYZE SELECT * FROM play_by_play "
            "WHERE season = 2023 AND posteam = 'KC' AND play_type = 'pass'"
        )).fetchall()
    plan_text = " ".join(str(row) for row in plan)
    assert "Seq Scan" not in plan_text, f"Unexpected Seq Scan: {plan_text[:200]}"

def test_odds_snapshot_write_read(pg_engine) -> None:
    with pg_engine.begin() as conn:
        conn.execute(sa.text(
            "INSERT INTO games (game_id, season, week, home_team, away_team, game_date) "
            "VALUES ('2023_01_KC_DET', 2023, 1, 'KC', 'DET', '2023-09-10') ON CONFLICT DO NOTHING"
        ))
        conn.execute(sa.text(
            "INSERT INTO odds_snapshots (game_id, sportsbook, market_type, line, price) "
            "VALUES ('2023_01_KC_DET', 'DraftKings', 'moneyline', NULL, -150)"
        ))
        row = conn.execute(sa.text(
            "SELECT game_id, sportsbook, market_type, price, snapped_at "
            "FROM odds_snapshots WHERE game_id='2023_01_KC_DET' ORDER BY snapped_at DESC LIMIT 1"
        )).fetchone()
    assert row is not None
    assert row[0] == "2023_01_KC_DET"
    assert row[1] == "DraftKings"
    assert row[3] == -150
    assert row[4] is not None  # snapped_at auto-populated
```
